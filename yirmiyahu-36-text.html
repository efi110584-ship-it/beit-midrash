הרקע לנבואה וכתיבת המגילה

א-ב בַּשָּׁנָה הָרְבִיעִית לִיהוֹיָקִים בֶּן יֹאשִׁיָּהוּ מֶלֶךְ יְהוּדָה, הָיָה הַדָּבָר הַזֶּה אֶל יִרְמְיָהוּ מֵאֵת ה' לֵאמֹר: קַח לְךָ מְגִלַּת סֵפֶר, וְכָתַבְתָּ אֵלֶיהָ אֵת כָּל הַדְּבָרִים אֲשֶׁר דִּבַּרְתִּי אֵלֶיךָ עַל יִשְׂרָאֵל וְעַל יְהוּדָה וְעַל כָּל הַגּוֹיִם, מִיּוֹם דַּבְּרִי אֵלֶיךָ, . ג , וְסָלַחְתִּי לַעֲוֹנָם וּלְחַטָּאתָם.

קריאת המגילה - הדרגתיות התגובות

ט-י וַיְהִי בַּשָּׁנָה הַחֲמִישִׁית לִיהוֹיָקִים בֶּן יֹאשִׁיָּהוּ מֶלֶךְ יְהוּדָה, בַּחֹדֶשׁ הַתְּשִׁעִי, קָרְאוּ צוֹם לִפְנֵי ה' כָּל הָעָם בִּירוּשָׁלִַם. וַיִּקְרָא בָרוּךְ בַּסֵּפֶר אֶת דִּבְרֵי יִרְמְיָהוּ בֵּית ה', בְּלִשְׁכַּת גְּמַרְיָהוּ בֶן שָׁפָן הַסֹּפֵר בֶּחָצֵר הָעֶלְיוֹן, . יא-יג וַיִּשְׁמַע מִכָיְהוּ בֶן גְּמַרְיָהוּ בֶן שָׁפָן אֵת כָּל דִּבְרֵי ה' מֵעַל הַסֵּפֶר, וַיֵּרֶד בֵּית הַמֶּלֶךְ, עַל לִשְׁכַּת הַסֹּפֵר, וְהִנֵּה שָׁם כָּל הַשָּׂרִים יוֹשְׁבִים. .

שריפת המגילה וכתיבתה מחדש

כג וַיְהִי כִּקְרוֹא יְהוּדִי שָׁלֹשׁ דְּלָתוֹת וְאַרְבָּעָה, , עַד תֹּם כָּל הַמְּגִלָּה עַל הָאֵשׁ אֲשֶׁר עַל הָאָח. כז-כח וַיְהִי דְבַר ה' אֶל יִרְמְיָהוּ, אַחֲרֵי שְׂרֹף הַמֶּלֶךְ אֶת הַמְּגִלָּה וְאֶת הַדְּבָרִים אֲשֶׁר כָּתַב בָּרוּךְ מִפִּי יִרְמְיָהוּ לֵאמֹר: מֶלֶךְ יְהוּדָה.
