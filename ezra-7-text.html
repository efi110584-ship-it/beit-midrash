יחוסו של עזרא

א, ה וְאַחַר הַדְּבָרִים הָאֵלֶּה, בְּמַלְכוּת אַרְתַּחְשַׁסְתְּא מֶלֶךְ פָּרָס: . ו הוּא עֶזְרָא עָלָה מִבָּבֶל, וְהוּא סֹפֵר מָהִיר בְּתוֹרַת מֹשֶׁה אֲשֶׁר נָתַן ה' אֱלֹקֵי יִשְׂרָאֵל, וַיִּתֶּן לוֹ הַמֶּלֶךְ .

עליית עזרא - תאריכים סמליים

ז וַיַּעֲלוּ מִבְּנֵי יִשְׂרָאֵל וּמִן הַכֹּהֲנִים וְהַלְוִיִּם וְהַמְשֹׁרְרִים וְהַשֹּׁעֲרִים וְהַנְּתִינִים אֶל יְרוּשָׁלַם, בִּשְׁנַת שֶׁבַע לְאַרְתַּחְשַׁסְתְּא הַמֶּלֶךְ. ח וַיָּבֹא יְרוּשָׁלַם בַּחֹדֶשׁ הַחֲמִישִׁי, הִיא שְׁנַת הַשְּׁבִיעִית לַמֶּלֶךְ. ט כִּי , כְּיַד אֱלֹקָיו הַטּוֹבָה עָלָיו. י כִּי עֶזְרָא הֵכִין לְבָבוֹ, לִדְרוֹשׁ אֶת תּוֹרַת ה', וְלַעֲשֹׂת וּלְלַמֵּד בְּיִשְׂרָאֵל חֹק וּמִשְׁפָּט.

הרישיון שקיבל עזרא

יא וְזֶה פַּרְשֶׁגֶן הַנִּשְׁתְּוָן אֲשֶׁר נָתַן הַמֶּלֶךְ אַרְתַּחְשַׁסְתְּא לְעֶזְרָא הַכֹּהֵן הַסֹּפֵר, סֹפֵר דִּבְרֵי מִצְוֹת ה' וְחֻקָּיו עַל יִשְׂרָאֵל. יב אַרְתַּחְשַׁסְתְּא מֶלֶךְ הַמְּלָכִים, לְעֶזְרָא הַכֹּהֵן, . יג מִמֶּנִּי הוּשַׂם צַו, שֶׁכָּל הַמִּתְנַדֵּב בְּמַלְכוּתִי מֵעַם יִשְׂרָאֵל וְכֹהֲנָיו וְהַלְוִיִּם לָלֶכֶת לִירוּשָׁלַיִם, עִמְּךָ יֵלֵךְ. יד כָּל זֹאת מִשּׁוּם שֶׁמִּלִּפְנֵי הַמֶּלֶךְ וְשִׁבְעַת יוֹעֲצָיו נִשְׁלַחְתָּ לְבַקֵּר עַל יְהוּדָה וִירוּשָׁלַיִם, בְּדַת אֱלֹקֶיךָ אֲשֶׁר בְּיָדֶךָ. טו וּלְהוֹבִיל אֶת הַכֶּסֶף וְהַזָּהָב אֲשֶׁר הַמֶּלֶךְ וְיוֹעֲצָיו הִתְנַדְּבוּ לֵאלֹקֵי יִשְׂרָאֵל, אֲשֶׁר בִּירוּשָׁלַיִם מִשְׁכָּנוֹ. טז וְכָל הַכֶּסֶף וְהַזָּהָב אֲשֶׁר תִּמְצָא בְּכָל מְדִינַת בָּבֶל, עִם הִתְנַדְּבוּת הָעָם וְהַכֹּהֲנִים הַמִּתְנַדְּבִים לְבֵית אֱלֹקֵיהֶם אֲשֶׁר בִּירוּשָׁלַיִם. יז לָכֵן בִּמְהִירוּת תִּקְנֶה בְּכֶסֶף זֶה פָּרִים אֵילִים כְּבָשִׂים, וּמִנְחוֹתֵיהֶם וְנִסְכֵּיהֶם, וְתַקְרִיב אוֹתָם עַל הַמִּזְבֵּחַ אֲשֶׁר בְּבֵית אֱלֹקֵיכֶם שֶׁבִּירוּשָׁלַיִם. יח וּמַה שֶּׁיִּיטַב בְּעֵינֶיךָ וּבְעֵינֵי אַחֶיךָ לַעֲשׂוֹת בִּשְׁאָר הַכֶּסֶף וְהַזָּהָב, כִּרְצוֹן אֱלֹקֵיכֶם תַּעֲשׂוּ. יט וְהַכֵּלִים אֲשֶׁר נִתְּנוּ לְךָ לַעֲבוֹדַת בֵּית אֱלֹקֶיךָ, מְסֹר לִפְנֵי אֱלֹקֵי יְרוּשָׁלָיִם. כ וּשְׁאָר צֹרֶךְ בֵּית אֱלֹקֶיךָ אֲשֶׁר יִפֹּל לְךָ לָתֵת, תִּתֵּן מִבֵּית גִּנְזֵי הַמֶּלֶךְ. כא וּמִמֶּנִּי, אֲנִי אַרְתַּחְשַׁסְתְּא הַמֶּלֶךְ, הוּשַׂם צַו לְכָל הַגִּזְבָּרִים אֲשֶׁר בְּעֵבֶר הַנָּהָר: כָּל אֲשֶׁר יִשְׁאַל מִכֶּם עֶזְרָא הַכֹּהֵן, סוֹפֵר דָּת אֱלֹקֵי הַשָּׁמַיִם, בִּמְהִירוּת יֵעָשֶׂה. כב עַד כֶּסֶף מֵאָה כִּכָּר, וְעַד חִטִּים מֵאָה כֹּר, וְעַד יַיִן מֵאָה בַּתִּים, וְעַד שֶׁמֶן מֵאָה בַּתִּים, וּמֶלַח לְלֹא הַגְבָּלָה. כג כָּל אֲשֶׁר בְּפִי אֱלֹקֵי הַשָּׁמַיִם, יֵעָשֶׂה בִּזְרִיזוּת לְבֵית אֱלֹקֵי הַשָּׁמַיִם, כִּי לָמָּה יִהְיֶה קֶצֶף עַל מַלְכוּת הַמֶּלֶךְ וּבָנָיו. כד וְלָכֶם מוֹדִיעִים, שֶׁכָּל הַכֹּהֲנִים וְהַלְוִיִּם הַמְשׁוֹרְרִים הַשּׁוֹעֲרִים הַנְּתִינִים וְעוֹבְדֵי בֵּית הָאֱלֹקִים הַזֶּה, אֵין רְשׁוּת לְהַטִּיל עֲלֵיהֶם מַס, מִנְחָה וְהֶלֶךְ. כה וְאַתָּה עֶזְרָא, כְּחָכְמַת אֱלֹקֶיךָ אֲשֶׁר בְּיָדְךָ, . כו וְכָל אֲשֶׁר לֹא יִהְיֶה עֹשֶׂה דָּת אֱלֹקֶיךָ וְדָת הַמֶּלֶךְ, בִּמְהִירוּת יֵעָשֶׂה בּוֹ דִין: אִם לְמָוֶת אִם לְגָלוּת אִם לְעֹנֶשׁ נְכָסִין וְלֶאֱסוּרִין.

תגובת עזרא - הכרת הטוב

כז-כח בָּרוּךְ ה' אֱלֹקֵי אֲבוֹתֵינוּ, אֲשֶׁר נָתַן כָּזֹאת בְּלֵב הַמֶּלֶךְ, לְפָאֵר אֶת בֵּית ה' אֲשֶׁר בִּירוּשָׁלָם. וְעָלַי הִטָּה חֶסֶד לִפְנֵי הַמֶּלֶךְ וְיוֹעֲצָיו... וַאֲנִי , וָאֶקְבְּצָה מִיִּשְׂרָאֵל .
