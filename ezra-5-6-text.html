חידוש הבנייה בעקבות נבואות חגי וזכריה

א-ב וְהִתְנַבְּאוּ עַל הַיְּהוּדִים אֲשֶׁר בִּיהוּדָה וּבִירוּשָׁלַיִם, בְּשֵׁם אֱלֹקֵי יִשְׂרָאֵל עֲלֵיהֶם. אָז קָמוּ זְרֻבָּבֶל בֶּן שְׁאַלְתִּיאֵל וְיֵשׁוּעַ בֶּן יוֹצָדָק, וְהֵחֵלּוּ לִבְנוֹת אֶת בֵּית הָאֱלֹקִים אֲשֶׁר בִּירוּשָׁלַיִם, וְעִמָּהֶם נְבִיאֵי הָאֱלֹקִים עוֹזְרִים לָהֶם.
הערה: פרק זה כתוב בארמית. במבחן הבגרות תידרשו לדעת את תוכן הפרק ולא את המילים בעל-פה.

נושא 2 מתוך 4
מציאת הצהרת כורש בגנזי המלך, ואישורה מחדש
עזרא ו', א'-י"ב (ארמית, מתורגם)

מציאת הצהרת כורש בגנזי המלך, ואישורה מחדש

ו-ז כָּעֵת, תַּתְּנַי פַּחַת עֵבֶר הַנָּהָר, שְׁתַר בּוֹזְנַי וְחַבְרֵיהֶם הָאֲפַרְסְכָיֵא אֲשֶׁר בְּעֵבֶר הַנָּהָר, מְרֻחָקִים הֱיוּ מִשָּׁם. . ח-י וּמִנִּי נִתַּן טַעַם לְמַה שֶּׁתַּעֲשׂוּ עִם זִקְנֵי הַיְּהוּדִים הָאֵלֶּה לִבְנוֹת אֶת בֵּית הָאֱלֹקִים הַזֶּה: וּמִנִּכְסֵי הַמֶּלֶךְ אֲשֶׁר מִדַּת עֵבֶר הַנָּהָר, , שֶׁלֹּא לְבַטֵּל. וּמַה שֶׁצָּרִיךְ - פָּרֵי בָקָר וְאֵילִים וּכְבָשִׂים לְעוֹלוֹת לֵאלֹקֵי הַשָּׁמַיִם, חִטִּים מֶלַח יַיִן וָשֶׁמֶן, כְּמַאֲמַר הַכֹּהֲנִים אֲשֶׁר בִּירוּשָׁלַיִם, יְהִי נִתָּן לָהֶם יוֹם בְּיוֹם בְּלֹא מִשְׁגֶּה, כְּדֵי שֶׁיִּהְיוּ מַקְרִיבִים נִיחוֹחִים לֵאלֹקֵי הַשָּׁמַיִם, וּמִתְפַּלְלִים לְחַיֵּי הַמֶּלֶךְ וּבָנָיו. יא-יב וּמִנִּי נִתַּן טַעַם, אֲשֶׁר כָּל אָדָם אֲשֶׁר יְשַׁנֶּה אֶת הַדָּבָר הַזֶּה, יִנָּתֵק עֵץ מִבֵּיתוֹ וְזָקוּף יֻכֶּה עָלָיו, וּבֵיתוֹ מַזְבֵּלָה יֵעָשֶׂה עַל זֹאת. וְהָאֱלֹקִים אֲשֶׁר שִׁכֵּן שְׁמוֹ שָׁם, יַפִּיל כָּל מֶלֶךְ וְעָם אֲשֶׁר יִשְׁלַח יָדוֹ לְהַשְׁנוֹת וּלְהַשְׁחִית אֶת בֵּית הָאֱלֹקִים הַזֶּה אֲשֶׁר בִּירוּשָׁלַיִם.

תיאור חנוכת המקדש

טו וְנִשְׁלַם בַּיִת זֶה עַד יוֹם שְׁלֹשָׁה לְחֹדֶשׁ אֲדָר, שֶׁהִיא שְׁנַת שֵׁשׁ לְמַלְכוּת דָּרְיָוֶשׁ הַמֶּלֶךְ. טז-יח וְעָשׂוּ בְנֵי יִשְׂרָאֵל, הַכֹּהֲנִים וְהַלְוִיִּים וּשְׁאָר בְּנֵי הַגּוֹלָה, . וְהִקְרִיבוּ לַחֲנֻכַּת בֵּית הָאֱלֹקִים הַזֶּה, פָּרִים מֵאָה, אֵילִים מָאתַיִם, כְּבָשִׂים אַרְבַּע מֵאוֹת, וּצְפִירֵי עִזִּים . וְהֶעֱמִידוּ הַכֹּהֲנִים בְּמַחְלְקוֹתֵיהֶם, וְהַלְוִיִּים בְּמִשְׁמְרוֹתֵיהֶם, עַל עֲבוֹדַת הָאֱלֹקִים אֲשֶׁר בִּירוּשָׁלַיִם, כִּכְתָב סֵפֶר מֹשֶׁה.
