חלום הסולם - התגלות ה' ליעקב

י-יב וַיֵּצֵא יַעֲקֹב מִבְּאֵר שָׁבַע וַיֵּלֶךְ חָרָנָה. וַיִּפְגַּע בַּמָּקוֹם וַיָּלֶן שָׁם כִּי בָא הַשֶּׁמֶשׁ, וַיִּקַּח מֵאַבְנֵי הַמָּקוֹם וַיָּשֶׂם מְרַאֲשֹׁתָיו וַיִּשְׁכַּב בַּמָּקוֹם הַהוּא. וַיַּחֲלֹם וְהִנֵּה סֻלָּם מֻצָּב אַרְצָה וְרֹאשׁוֹ מַגִּיעַ הַשָּׁמָיְמָה, וְהִנֵּה מַלְאֲכֵי אֱלֹהִים בּוֹ. יג וְהִנֵּה ה' נִצָּב עָלָיו וַיֹּאמַר: אֲנִי ה' אֱלֹהֵי אַבְרָהָם אָבִיךָ וֵאלֹהֵי יִצְחָק, לְךָ אֶתְּנֶנָּה וּלְזַרְעֶךָ.

הבטחת השמירה, והתעוררות יעקב

יד-טו וְהָיָה זַרְעֲךָ כַּעֲפַר הָאָרֶץ... וְנִבְרְכוּ בְךָ כָּל מִשְׁפְּחֹת הָאֲדָמָה וּבְזַרְעֶךָ. וְהִנֵּה אָנֹכִי עִמָּךְ בְּכֹל אֲשֶׁר תֵּלֵךְ, וַהֲשִׁבֹתִיךָ אֶל הָאֲדָמָה הַזֹּאת. טז-יז וַיִּיקַץ יַעֲקֹב מִשְּׁנָתוֹ וַיֹּאמֶר: אָכֵן יֵשׁ ה' בַּמָּקוֹם הַזֶּה . וַיִּירָא וַיֹּאמַר: מַה נּוֹרָא הַמָּקוֹם הַזֶּה! אֵין זֶה כִּי אִם בֵּית אֱלֹהִים וְזֶה שַׁעַר הַשָּׁמָיִם.

קריאת שם המקום ונדר יעקב

יח-יט וַיַּשְׁכֵּם יַעֲקֹב בַּבֹּקֶר וַיִּקַּח אֶת הָאֶבֶן... וַיָּשֶׂם אֹתָהּ מַצֵּבָה, וַיִּצֹק שֶׁמֶן עַל רֹאשָׁהּ. וַיִּקְרָא אֶת שֵׁם הַמָּקוֹם הַהוּא בֵּית אֵל. כ-כב וַיִּדַּר יַעֲקֹב נֶדֶר לֵאמֹר: , וּשְׁמָרַנִי בַּדֶּרֶךְ הַזֶּה אֲשֶׁר אָנֹכִי הוֹלֵךְ, וְנָתַן לִי לֶחֶם לֶאֱכֹל וּבֶגֶד לִלְבֹּשׁ, וְשַׁבְתִּי בְשָׁלוֹם אֶל בֵּית אָבִי, . וְהָאֶבֶן הַזֹּאת אֲשֶׁר שַׂמְתִּי מַצֵּבָה יִהְיֶה בֵּית אֱלֹהִים, וְכֹל אֲשֶׁר תִּתֶּן לִי עַשֵּׂר אֲעַשְּׂרֶנּוּ לָךְ.
