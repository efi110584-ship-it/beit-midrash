נושא 1

🎯 תכניתו של יצחק ותכניתה של רבקה

פרק כ"ז, פסוקים א'-י"ז

1 קוראים את הפסוקים

⭐ פסוק מרכזי - חשוב במיוחד להבנה ולבגרות

א'

וַיְהִי כִּי זָקֵן יִצְחָק וַתִּכְהֶיןָ עֵינָיו מֵרְאֹת וַיִּקְרָא אֶת עֵשָׂו בְּנוֹ הַגָּדֹל וַיֹּאמֶר אֵלָיו בְּנִי וַיֹּאמֶר אֵלָיו הִנֵּנִי

כשיצחק הזקין ועיניו נחלשו ולא ראה, הוא קרא לעשו בנו הגדול ואמר לו: בני; ועשו ענה: הנני

ב'

וַיֹּאמֶר הִנֵּה נָא זָקַנְתִּי לֹא יָדַעְתִּי יוֹם מוֹתִי

יצחק אמר: הנה הזקנתי, ואיני יודע מתי אמות

ג'

וְעַתָּה שָׂא נָא כֵלֶיךָ תֶּלְיְךָ וְקַשְׁתֶּךָ וְצֵא הַשָּׂדֶה וְצוּדָה לִּי צָיִד

ועכשיו קח את כלי הציד שלך - את אשפת החצים ואת הקשת - צא לשדה וצוד לי ציד

ד'⭐

וַעֲשֵׂה לִי מַטְעַמִּים כַּאֲשֶׁר אָהַבְתִּי וְהָבִיאָה לִּי וְאֹכֵלָה בַּעֲבוּר תְּבָרֶכְךָ נַפְשִׁי בְּטֶרֶם אָמוּת

הכן לי מאכלים טעימים כמו שאני אוהב, והבא לי ואוכל - כדי שאברך אותך לפני שאמות

ה'

וְרִבְקָה שֹׁמַעַת בְּדַבֵּר יִצְחָק אֶל עֵשָׂו בְּנוֹ וַיֵּלֶךְ עֵשָׂו הַשָּׂדֶה לָצוּד צַיִד לְהָבִיא

רבקה שמעה כשיצחק דיבר אל עשו בנו; ועשו הלך לשדה לצוד ציד ולהביא

ו'

וְרִבְקָה אָמְרָה אֶל יַעֲקֹב בְּנָהּ לֵאמֹר הִנֵּה שָׁמַעְתִּי אֶת אָבִיךָ מְדַבֵּר אֶל עֵשָׂו אָחִיךָ לֵאמֹר

רבקה אמרה ליעקב בנה: הנה שמעתי את אביך מדבר אל עשו אחיך ואומר:

ז'

הָבִיאָה לִּי צַיִד וַעֲשֵׂה לִי מַטְעַמִּים וְאֹכֵלָה וַאֲבָרֶכְכָה לִפְנֵי ה' לִפְנֵי מוֹתִי

הבא לי ציד והכן לי מאכלים טעימים, ואוכל ואברך אותך לפני ה' לפני מותי

ח'

וְעַתָּה בְנִי שְׁמַע בְּקֹלִי לַאֲשֶׁר אֲנִי מְצַוָּה אֹתָךְ

ועכשיו, בני, שמע בקולי - למה שאני מצווה אותך

ט'

לֶךְ נָא אֶל הַצֹּאן וְקַח לִי מִשָּׁם שְׁנֵי גְּדָיֵי עִזִּים טֹבִים וְאֶעֱשֶׂה אֹתָם מַטְעַמִּים לְאָבִיךָ כַּאֲשֶׁר אָהֵב

לך אל הצאן וקח לי משם שני גדיי עזים טובים, ואכין מהם מאכלים טעימים לאביך, כמו שהוא אוהב

י'

וְהֵבֵאתָ לְאָבִיךָ וְאָכָל בַּעֲבֻר אֲשֶׁר יְבָרֶכְךָ לִפְנֵי מוֹתוֹ

ותביא לאביך והוא יאכל, כדי שיברך אותך לפני מותו

י"א

וַיֹּאמֶר יַעֲקֹב אֶל רִבְקָה אִמּוֹ הֵן עֵשָׂו אָחִי אִישׁ שָׂעִר וְאָנֹכִי אִישׁ חָלָק

יעקב אמר לרבקה אמו: הרי עשו אחי איש שעיר, ואני איש חלק

י"ב

אוּלַי יְמֻשֵּׁנִי אָבִי וְהָיִיתִי בְעֵינָיו כִּמְתַעְתֵּעַ וְהֵבֵאתִי עָלַי קְלָלָה וְלֹא בְרָכָה

אולי אבי ימשש אותי, ואהיה בעיניו כמו רמאי - ואביא על עצמי קללה ולא ברכה

י"ג⭐

וַתֹּאמֶר לוֹ אִמּוֹ עָלַי קִלְלָתְךָ בְּנִי אַךְ שְׁמַע בְּקֹלִי וְלֵךְ קַח לִי

אמו אמרה לו: הקללה שלך תהיה עליי, בני; רק שמע בקולי ולך קח לי

י"ד

וַיֵּלֶךְ וַיִּקַּח וַיָּבֵא לְאִמּוֹ וַתַּעַשׂ אִמּוֹ מַטְעַמִּים כַּאֲשֶׁר אָהֵב אָבִיו

יעקב הלך, לקח והביא לאמו; ואמו הכינה מאכלים טעימים כמו שאביו אוהב

ט"ו

וַתִּקַּח רִבְקָה אֶת בִּגְדֵי עֵשָׂו בְּנָהּ הַגָּדֹל הַחֲמֻדֹת אֲשֶׁר אִתָּהּ בַּבָּיִת וַתַּלְבֵּשׁ אֶת יַעֲקֹב בְּנָהּ הַקָּטָן

רבקה לקחה את הבגדים היפים של עשו בנה הגדול, שהיו אצלה בבית, והלבישה את יעקב בנה הקטן

ט"ז

וְאֵת עֹרֹת גְּדָיֵי הָעִזִּים הִלְבִּישָׁה עַל יָדָיו וְעַל חֶלְקַת צַוָּארָיו

ואת עורות גדיי העזים הלבישה על ידיו ועל החלק החלק של צווארו

י"ז

וַתִּתֵּן אֶת הַמַּטְעַמִּים וְאֶת הַלֶּחֶם אֲשֶׁר עָשָׂתָה בְּיַד יַעֲקֹב בְּנָהּ

ונתנה את המאכלים הטעימים ואת הלחם שהכינה ביד יעקב בנה

3 הפרשנים - לומדים ביחד

בראשית רבהפסוק י"ד - "וַיֵּלֶךְ וַיִּקַּח וַיָּבֵא"
וילך ויקח ויבא לאמו - אנוס וכפוף ובוכה

פירוש שלמדתםרמב"ן / רד"ק / הרש"ר הירש - פסוקים ד'-ה'

רמב"ן

(ד) היה בדעתו לברך אותו [את עשו] שיזכה הוא בברכת אברהם לנחול את הארץ, ולהיות הוא בעל הברית לאלוהים כי הוא הבכור: (טו) ...כי המנהג במולידים להכיר את הבכור בברכה ובכבוד ובמתן. (כא) ...כי לא היה עשו רשע בעיני אביו. (טו) והיא, מדעתה צִדקת הקטן ורשעת הגדול תשתדל בכל הטורח הזה להעביר הברכה והכבוד מן הגדול אל הקטן.

רד"ק

(ד) ועשה לי מטעמים ... - לפיכך שאל יצחק מעשו בנו להביא ציד וייטב לבו בו ויברכנו, כי ידע כי צריך הוא לברכתו כי לא היה איש הגון וטוב. אבל יעקב לא היה צריך לברכה, כי ידע יצחק כי ברכת אברהם לו תהיה וברית קיום הזרע המיוחד לאברהם, ובניו ירשו את הארץ. ואברהם גם כן לא ברך את יצחק. (ה) ורבקה שמעת בדבר יצחק - וראתה כי הלך עשו לעשות מצות אביו ואמרה ליעקב, כי קנאה בזה. והיא לא ידעה כי יעקב יהיה מבורך אפילו לא יברכנהו אביו.

הרש"ר הירש

שני בניו של יצחק ייצגו שני יסודות בביתו, עשיו את הכוח החומרי, יעקב את הכוח הרוחני. יודע היה יצחק כי האומה זקוקה לשני הכיוונים... לפיכך הועיד לעשיו ברכה בעלת תוכן חומרי, וברכה רוחנית היתה שמורה עמו ליעקב... אולם מבית לבן ידעה רבקה את הקללה שבפיצול זה... החומר שאין עמו רוח, היה בעיניה קללה, את הברכה ראתה רק בלתי מחולקת, יורדת על ראשו של אחד.

נושא 2

🎭 יעקב לפני יצחק, ועשו חוזר

פרק כ"ז, פסוקים י"ח-מ'

1 קוראים את הפסוקים

⭐ פסוק מרכזי - חשוב במיוחד להבנה ולבגרות

י"ח

וַיָּבֹא אֶל אָבִיו וַיֹּאמֶר אָבִי וַיֹּאמֶר הִנֶּנִּי מִי אַתָּה בְּנִי

יעקב בא אל אביו ואמר: אבי; יצחק ענה: הנני; מי אתה, בני?

י"ט⭐

וַיֹּאמֶר יַעֲקֹב אֶל אָבִיו אָנֹכִי עֵשָׂו בְּכֹרֶךָ עָשִׂיתִי כַּאֲשֶׁר דִּבַּרְתָּ אֵלָי קוּם נָא שְׁבָה וְאָכְלָה מִצֵּידִי בַּעֲבוּר תְּבָרֲכַנִּי נַפְשֶׁךָ

יעקב אמר לאביו: אנכי עשו בכורך; עשיתי כמו שאמרת לי; קום נא, שב ואכול מן הציד שלי, כדי שתברך אותי

כ'

וַיֹּאמֶר יִצְחָק אֶל בְּנוֹ מַה זֶּה מִהַרְתָּ לִמְצֹא בְּנִי וַיֹּאמֶר כִּי הִקְרָה ה' אֱלֹהֶיךָ לְפָנָי

יצחק אמר לבנו: איך מיהרת כל כך למצוא, בני? ויעקב אמר: כי ה' אלוהיך זימן (הביא) לפניי

כ"א

וַיֹּאמֶר יִצְחָק אֶל יַעֲקֹב גְּשָׁה נָּא וַאֲמֻשְׁךָ בְּנִי הַאַתָּה זֶה בְּנִי עֵשָׂו אִם לֹא

יצחק אמר ליעקב: גש נא ואמשש אותך, בני - האם אתה בני עשו או לא?

כ"ב⭐

וַיִּגַּשׁ יַעֲקֹב אֶל יִצְחָק אָבִיו וַיְמֻשֵּׁהוּ וַיֹּאמֶר הַקֹּל קוֹל יַעֲקֹב וְהַיָּדַיִם יְדֵי עֵשָׂו

יעקב ניגש אל יצחק אביו, ויצחק מישש אותו ואמר: הקול - קול יעקב, והידיים - ידי עשו

כ"ג

וְלֹא הִכִּירוֹ כִּי הָיוּ יָדָיו כִּידֵי עֵשָׂו אָחִיו שְׂעִרֹת וַיְבָרְכֵהוּ

יצחק לא זיהה אותו, כי ידיו היו שעירות כמו ידי עשו אחיו - ובירך אותו

כ"ד

וַיֹּאמֶר אַתָּה זֶה בְּנִי עֵשָׂו וַיֹּאמֶר אָנִי

יצחק אמר: האם אתה בני עשו? ויעקב אמר: אני

כ"ה

וַיֹּאמֶר הַגִּשָׁה לִּי וְאֹכְלָה מִצֵּיד בְּנִי לְמַעַן תְּבָרֶכְךָ נַפְשִׁי וַיַּגֶּשׁ לוֹ וַיֹּאכַל וַיָּבֵא לוֹ יַיִן וַיֵּשְׁתְּ

יצחק אמר: הגש לי ואוכל מן הציד של בני, כדי שאברך אותך; יעקב הגיש לו והוא אכל, הביא לו יין והוא שתה

כ"ו

וַיֹּאמֶר אֵלָיו יִצְחָק אָבִיו גְּשָׁה נָּא וּשְׁקָה לִּי בְּנִי

יצחק אביו אמר לו: גש נא ונשק לי, בני

כ"ז

וַיִּגַּשׁ וַיִּשַּׁק לוֹ וַיָּרַח אֶת רֵיחַ בְּגָדָיו וַיְבָרֲכֵהוּ וַיֹּאמֶר רְאֵה רֵיחַ בְּנִי כְּרֵיחַ שָׂדֶה אֲשֶׁר בֵּרֲכוֹ ה'

יעקב ניגש ונישק לו; יצחק הריח את ריח בגדיו, בירך אותו ואמר: ראה, ריח בני כמו ריח שדה שה' בירך

כ"ח

וְיִתֶּן לְךָ הָאֱלֹהִים מִטַּל הַשָּׁמַיִם וּמִשְׁמַנֵּי הָאָרֶץ וְרֹב דָּגָן וְתִירֹשׁ

ייתן לך האלוהים מטל השמיים ומן האדמה השמנה (הפורייה), ושפע של תבואה ויין

כ"ט⭐

יַעַבְדוּךָ עַמִּים וְיִשְׁתַּחֲווּ לְךָ לְאֻמִּים הֱוֵה גְבִיר לְאַחֶיךָ וְיִשְׁתַּחֲוּוּ לְךָ בְּנֵי אִמֶּךָ אֹרְרֶיךָ אָרוּר וּמְבָרֲכֶיךָ בָּרוּךְ

עמים יעבדו אותך ולאומים ישתחוו לך; היה אדון לאחיך, ובני אמך ישתחוו לך; מי שמקלל אותך - ארור, ומי שמברך אותך - ברוך

ל'

וַיְהִי כַּאֲשֶׁר כִּלָּה יִצְחָק לְבָרֵךְ אֶת יַעֲקֹב וַיְהִי אַךְ יָצֹא יָצָא יַעֲקֹב מֵאֵת פְּנֵי יִצְחָק אָבִיו וְעֵשָׂו אָחִיו בָּא מִצֵּידוֹ

כשיצחק גמר לברך את יעקב, ויעקב רק יצא מלפני יצחק אביו - עשו אחיו בא מן הציד שלו.

ל"א

וַיַּעַשׂ גַּם הוּא מַטְעַמִּים וַיָּבֵא לְאָבִיו וַיֹּאמֶר לְאָבִיו יָקֻם אָבִי וְיֹאכַל מִצֵּיד בְּנוֹ בַּעֲבוּר תְּבָרֲכַנִּי נַפְשֶׁךָ

גם עשו הכין מאכלים טעימים והביא לאביו, ואמר לו: שאבי יקום ויאכל מן הציד של בנו, כדי שתברך אותי.

ל"ב

וַיֹּאמֶר לוֹ יִצְחָק אָבִיו מִי אָתָּה וַיֹּאמֶר אֲנִי בִּנְךָ בְכֹרְךָ עֵשָׂו

יצחק אביו אמר לו: מי אתה? ועשו אמר: אני בנך, הבכור שלך, עשו.

ל"ג⭐

וַיֶּחֱרַד יִצְחָק חֲרָדָה גְּדֹלָה עַד מְאֹד וַיֹּאמֶר מִי אֵפוֹא הוּא הַצָּד צַיִד וַיָּבֵא לִי וָאֹכַל מִכֹּל בְּטֶרֶם תָּבוֹא וָאֲבָרֲכֵהוּ גַּם בָּרוּךְ יִהְיֶה

יצחק נבהל בהלה גדולה מאוד, ואמר: מי, אם כן, הוא זה שצד ציד והביא לי, ואכלתי מכול לפני שבאת, ובירכתי אותו? גם ברוך יהיה!

ל"ד

כִּשְׁמֹעַ עֵשָׂו אֶת דִּבְרֵי אָבִיו וַיִּצְעַק צְעָקָה גְּדֹלָה וּמָרָה עַד מְאֹד וַיֹּאמֶר לְאָבִיו בָּרֲכֵנִי גַם אָנִי אָבִי

כשעשו שמע את דברי אביו, צעק צעקה גדולה ומרה מאוד, ואמר לאביו: ברך גם אותי, אבי!

ל"ה

יצחק אמר: אחיך בא בעורמה ולקח את הברכה שלך

ל"ו⭐

וַיֹּאמֶר הֲכִי קָרָא שְׁמוֹ יַעֲקֹב וַיַּעְקְבֵנִי זֶה פַעֲמַיִם אֶת בְּכֹרָתִי לָקָח וְהִנֵּה עַתָּה לָקַח בִּרְכָתִי וַיֹּאמַר הֲלֹא אָצַלְתָּ לִּי בְּרָכָה

עשו אמר: האם לא בגלל זה קראו לו יעקב? הוא רימה אותי כבר פעמיים - את הבכורה שלי לקח, והנה עכשיו לקח את הברכה שלי! ואמר: האם לא השארת לי ברכה?

ל"ז

וַיַּעַן יִצְחָק וַיֹּאמֶר לְעֵשָׂו הֵן גְּבִיר שַׂמְתִּיו לָךְ וְאֶת כָּל אֶחָיו נָתַתִּי לוֹ לַעֲבָדִים וְדָגָן וְתִירֹשׁ סְמַכְתִּיו וּלְכָה אֵפוֹא מָה אֶעֱשֶׂה בְּנִי

יצחק ענה לעשו: הנה עשיתי אותו אדון עליך, ואת כל אחיו נתתי לו לעבדים, ונתתי לו דגן ויין; ולך - מה אוכל לעשות, בני?

ל"ח

וַיֹּאמֶר עֵשָׂו אֶל אָבִיו הַבְרָכָה אַחַת הִוא לְךָ אָבִי בָּרֲכֵנִי גַם אָנִי אָבִי וַיִּשָּׂא עֵשָׂו קֹלוֹ וַיֵּבְךְּ

עשו אמר לאביו: האם יש לך רק ברכה אחת, אבי? ברך גם אותי, אבי! ועשו בכה בקול.

ל"ט

וַיַּעַן יִצְחָק אָבִיו וַיֹּאמֶר אֵלָיו הִנֵּה מִשְׁמַנֵּי הָאָרֶץ יִהְיֶה מוֹשָׁבֶךָ וּמִטַּל הַשָּׁמַיִם מֵעָל

יצחק אביו ענה ואמר לו: הנה מקום מושבך יהיה באדמה שמנה (פורייה), ומטל השמיים מלמעלה

מ'⭐

וְעַל חַרְבְּךָ תִחְיֶה וְאֶת אָחִיךָ תַּעֲבֹד וְהָיָה כַּאֲשֶׁר תָּרִיד וּפָרַקְתָּ עֻלּוֹ מֵעַל צַוָּארֶךָ

על חרבך תחיה, ואת אחיך תעבוד; וכאשר תתמרד (כשאחיך יחטא) - תפרוק את העול שלו מעל צווארך
