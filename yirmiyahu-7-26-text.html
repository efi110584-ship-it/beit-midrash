נבואת ירמיהו על היכל ה' - האזהרה

ד אַל תִּבְטְחוּ לָכֶם אֶל דִּבְרֵי הַשֶּׁקֶר לֵאמֹר: . ח-ט הִנֵּה אַתֶּם בֹּטְחִים לָכֶם עַל דִּבְרֵי הַשֶּׁקֶר לְבִלְתִּי הוֹעִיל. הֲגָנֹב רָצֹחַ וְנָאֹף וְהִשָּׁבֵעַ לַשֶּׁקֶר וְקַטֵּר לַבַּעַל, וְהָלֹךְ אַחֲרֵי אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא יְדַעְתֶּם. י-יא וּבָאתֶם וַעֲמַדְתֶּם לְפָנַי בַּבַּיִת הַזֶּה אֲשֶׁר נִקְרָא שְׁמִי עָלָיו, וַאֲמַרְתֶּם נִצַּלְנוּ, לְמַעַן עֲשׂוֹת אֵת כָּל הַתּוֹעֵבוֹת הָאֵלֶּה. ?

תיקון המעשים ותקדים שילה

ה-ז כִּי אִם הֵיטֵיב תֵּיטִיבוּ אֶת דַּרְכֵיכֶם וְאֶת מַעַלְלֵיכֶם, . וְשִׁכַּנְתִּי אֶתְכֶם בַּמָּקוֹם הַזֶּה, בָּאָרֶץ אֲשֶׁר נָתַתִּי לַאֲבוֹתֵיכֶם לְמִן עוֹלָם וְעַד עוֹלָם. יב כִּי לְכוּ נָא אֶל מְקוֹמִי אֲשֶׁר בְּשִׁילוֹ, אֲשֶׁר שִׁכַּנְתִּי שְׁמִי שָׁם בָּרִאשׁוֹנָה, .

משפט ירמיהו - תקדימים סותרים

ח-ט וַיְהִי כְּכַלּוֹת יִרְמְיָהוּ לְדַבֵּר אֵת כָּל אֲשֶׁר צִוָּה ה' לְדַבֵּר אֶל כָּל הָעָם, וַיִּתְפְּשׂוּ אֹתוֹ הַכֹּהֲנִים וְהַנְּבִאִים וְכָל הָעָם לֵאמֹר: מוֹת תָּמוּת. טז-יז וַיֹּאמְרוּ הַשָּׂרִים וְכָל הָעָם אֶל הַכֹּהֲנִים וְאֶל הַנְּבִאִים: אֵין לָאִישׁ הַזֶּה מִשְׁפַּט מָוֶת, כִּי בְּשֵׁם ה' אֱלֹהֵינוּ דִּבֶּר אֵלֵינוּ. וַיָּקֻמוּ אֲנָשִׁים מִזִּקְנֵי הָאָרֶץ, וַיֹּאמְרוּ אֶל כָּל קְהַל הָעָם לֵאמֹר. יט . כ-כא וְגַם אִישׁ הָיָה מִתְנַבֵּא בְּשֵׁם ה', אוּרִיָּהוּ בֶּן שְׁמַעְיָהוּ מִקִּרְיַת הַיְּעָרִים, וַיִּנָּבֵא עַל הָעִיר הַזֹּאת וְעַל הָאָרֶץ הַזֹּאת כְּכֹל דִּבְרֵי יִרְמְיָהוּ. ; וַיִּשְׁמַע אוּרִיָּהוּ וַיִּרָא, וַיִּבְרַח וַיָּבֹא מִצְרָיִם. כג וַיּוֹצִיאוּ אֶת אוּרִיָּהוּ מִמִּצְרַיִם וַיְבִאֻהוּ אֶל הַמֶּלֶךְ יְהוֹיָקִים, וַיַּכֵּהוּ בֶּחָרֶב, וַיַּשְׁלֵךְ אֶת נִבְלָתוֹ אֶל קִבְרֵי בְּנֵי הָעָם.
